import React, { useState } from 'react';
import { Layers, Download, Upload, Check, Copy, Share2, X, RefreshCw, Smartphone } from 'lucide-react';
import { Slab } from '../types';
import { vaultAudio } from '../utils/audio';

interface CloudSyncModalProps {
  isOpen: boolean;
  onClose: () => void;
  slabs: Slab[];
  onImportSlabs: (importedSlabs: Slab[]) => void;
}

export const CloudSyncModal: React.FC<CloudSyncModalProps> = ({
  isOpen,
  onClose,
  slabs,
  onImportSlabs,
}) => {
  const [copiedLink, setCopiedLink] = useState(false);
  const [syncStatus, setSyncStatus] = useState<'idle' | 'syncing' | 'success'>('idle');

  if (!isOpen) return null;

  const handleExportJSON = () => {
    vaultAudio.playButtonTick();
    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(slabs, null, 2));
    const downloadAnchor = document.createElement('a');
    downloadAnchor.setAttribute("href", dataStr);
    downloadAnchor.setAttribute("download", `slabvault-backup-${new Date().toISOString().slice(0, 10)}.json`);
    document.body.appendChild(downloadAnchor);
    downloadAnchor.click();
    downloadAnchor.remove();
  };

  const handleImportFile = (e: React.ChangeEvent<HTMLInputElement>) => {
    const fileReader = new FileReader();
    if (e.target.files && e.target.files[0]) {
      fileReader.readAsText(e.target.files[0], "UTF-8");
      fileReader.onload = (event) => {
        try {
          const parsed = JSON.parse(event.target?.result as string);
          if (Array.isArray(parsed)) {
            onImportSlabs(parsed);
            vaultAudio.playSuccessFanfare();
            alert(`Successfully restored ${parsed.length} slabs to your vault!`);
            onClose();
          }
        } catch (err) {
          alert('Invalid backup file. Please select a valid SlabVault JSON backup.');
        }
      };
    }
  };

  const handleCopyShareLink = () => {
    navigator.clipboard.writeText(window.location.href);
    setCopiedLink(true);
    vaultAudio.playButtonTick();
    setTimeout(() => setCopiedLink(false), 2000);
  };

  return (
    <div className="fixed inset-0 z-50 bg-black/85 backdrop-blur-md flex items-center justify-center p-4">
      <div className="bg-zinc-950 border border-cyan-500/40 rounded-2xl max-w-lg w-full p-6 space-y-6 shadow-[0_0_50px_rgba(0,240,255,0.15)] relative">
        <button
          onClick={() => {
            vaultAudio.playButtonTick();
            onClose();
          }}
          className="absolute top-5 right-5 text-zinc-400 hover:text-white p-1 cursor-pointer"
        >
          <X className="w-5 h-5" />
        </button>

        <div className="flex items-center space-x-3">
          <div className="w-10 h-10 rounded-xl bg-cyan-500/20 border border-cyan-400/40 flex items-center justify-center">
            <Layers className="w-5 h-5 text-cyan-400" />
          </div>
          <div>
            <h2 className="text-xl font-bold text-white">Vault Backup & Cross-Device Sync</h2>
            <p className="text-xs text-zinc-400">Keep your slabs synced across iPhone, Android, and Desktop.</p>
          </div>
        </div>

        <div className="space-y-3">
          {/* Export Backup */}
          <div className="p-4 rounded-xl bg-zinc-900/60 border border-zinc-800 space-y-2">
            <div className="flex items-center justify-between">
              <div>
                <h4 className="text-sm font-bold text-white">Export Master Vault JSON</h4>
                <p className="text-xs text-zinc-400">Download a full offline snapshot of all {slabs.length} slabs & high-res cards.</p>
              </div>
              <button
                onClick={handleExportJSON}
                className="px-3.5 py-2 rounded-xl bg-cyan-500/20 hover:bg-cyan-500/30 border border-cyan-400/40 text-cyan-300 font-bold text-xs flex items-center space-x-1.5 transition-colors cursor-pointer"
              >
                <Download className="w-4 h-4" />
                <span>Export</span>
              </button>
            </div>
          </div>

          {/* Import Restore */}
          <div className="p-4 rounded-xl bg-zinc-900/60 border border-zinc-800 space-y-2">
            <div className="flex items-center justify-between">
              <div>
                <h4 className="text-sm font-bold text-white">Restore / Import Vault</h4>
                <p className="text-xs text-zinc-400">Load a previous SlabVault backup file from your device.</p>
              </div>
              <label className="px-3.5 py-2 rounded-xl bg-zinc-800 hover:bg-zinc-700 border border-zinc-700 text-zinc-200 font-bold text-xs flex items-center space-x-1.5 transition-colors cursor-pointer">
                <Upload className="w-4 h-4" />
                <span>Import JSON</span>
                <input
                  type="file"
                  accept=".json"
                  onChange={handleImportFile}
                  className="hidden"
                />
              </label>
            </div>
          </div>

          {/* Direct URL Share */}
          <div className="p-4 rounded-xl bg-zinc-900/60 border border-zinc-800 space-y-2">
            <div className="flex items-center justify-between">
              <div>
                <h4 className="text-sm font-bold text-white">Share Vault Link</h4>
                <p className="text-xs text-zinc-400">Open your portfolio instantly on mobile devices.</p>
              </div>
              <button
                onClick={handleCopyShareLink}
                className="px-3.5 py-2 rounded-xl bg-amber-500/20 hover:bg-amber-500/30 border border-amber-400/40 text-amber-300 font-bold text-xs flex items-center space-x-1.5 transition-colors cursor-pointer"
              >
                {copiedLink ? <Check className="w-4 h-4 text-emerald-400" /> : <Copy className="w-4 h-4" />}
                <span>{copiedLink ? 'Copied!' : 'Copy Link'}</span>
              </button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};
